[ ] 2.2 Complete Route Code to Paste into `backend/controllers/expense_controller.py`:

```python
# =====================================================================
# DEDICATED EXPENSE SHEET PDF DATA ENDPOINT
# GET /api/expenses/<int:expense_id>/pdf-data
# =====================================================================
@expense_bp.route("/<int:expense_id>/pdf-data", methods=["GET"])
@token_required
def get_expense_pdf_data(current_user_id, expense_id):
    """
    Fetches fully-resolved relational data for the dynamic Expense Sheet PDF:
    - Payer name and live UPI ID (for QR code & payment box)
    - Full split distribution member list with Payer / Participant roles
    - Attached bill / receipt image URLs
    - Precise timestamps: expenseDate, createdAt, updatedAt
    - Dynamic settlement status: COMPLETED if all participants are settled, else PENDING
    """
    async def _fetch():
        db = Prisma()
        await db.connect()
        try:
            # 1. Fetch expense with all required nested relations
            exp = await db.expense.find_unique(
                where={"id": expense_id},
                include={
                    "category": True,
                    "paidBy": True,
                    "createdBy": True,
                    "participants": {
                        "include": {
                            "user": True
                        }
                    }
                }
            )

            # 2. Return 404 if expense doesn't exist
            if not exp:
                return jsonify({
                    "success": False,
                    "error": f"Expense #{expense_id} not found"
                }), 404

            # 3. Format and serialize split distribution participants
            participants_list = []
            for p in (exp.participants or []):
                user_obj = getattr(p, "user", None)
                is_payer = (p.userId == exp.paidById)
                
                participants_list.append({
                    "userId": p.userId,
                    "name": user_obj.name if user_obj else f"User #{p.userId}",
                    "email": getattr(user_obj, "email", None),
                    "upiId": getattr(user_obj, "upi_id", None),
                    "shareAmount": float(p.shareAmount) if p.shareAmount is not None else 0.0,
                    "isPayer": is_payer,
                    "role": "Payer" if is_payer else "Participant",
                    "isSettled": is_payer or bool(p.isSettled) or (exp.status == "COMPLETED"),
                    "settledAt": p.settledAt.isoformat() if getattr(p, "settledAt", None) else None
                })

            # Sort participants: Payer first, then alphabetically by name
            participants_list.sort(key=lambda x: (not x["isPayer"], x["name"].lower()))

            # 4. Compute live settlement status
            non_payers = [p for p in (exp.participants or []) if p.userId != exp.paidById]
            is_all_settled = len(non_payers) == 0 or all(bool(p.isSettled) for p in non_payers)
            computed_status = "COMPLETED" if is_all_settled else "PENDING"

            # 5. Build clean, unified payload for React PDF Generator
            pdf_payload = {
                "id": exp.id,
                "title": exp.title,
                "description": exp.description or "",
                "amount": float(exp.amount),
                "currency": getattr(exp, "currency", "INR") or "INR",
                "status": computed_status,
                "expenseDate": exp.expenseDate.isoformat() if exp.expenseDate else None,
                "createdAt": exp.createdAt.isoformat() if exp.createdAt else None,
                "updatedAt": exp.updatedAt.isoformat() if exp.updatedAt else None,
                
                # Category details
                "categoryId": exp.categoryId,
                "categoryName": exp.category.name if exp.category else "General",
                
                # Payer details (Payment settlement target)
                "paidById": exp.paidById,
                "paidByName": exp.paidBy.name if exp.paidBy else "Unknown",
                "paidByUpiId": getattr(exp.paidBy, "upi_id", None) if getattr(exp, "paidBy", None) else None,
                
                # Recorder details
                "createdById": exp.createdById,
                "createdByName": exp.createdBy.name if exp.createdBy else "Unknown",
                
                # Attached Bill & Receipt Image URLs
                "images": getattr(exp, "images", []) or [],
                
                # Split Distribution List
                "participants": participants_list,
                "participantCount": len(participants_list)
            }

            return jsonify({
                "success": True,
                "data": pdf_payload
            }), 200

        except Exception as e:
            return jsonify({
                "success": False,
                "error": f"Failed to retrieve expense PDF data: {str(e)}"
            }), 500
        finally:
            if db.is_connected():
                await db.disconnect()

    import asyncio
    return asyncio.run(_fetch())
```
