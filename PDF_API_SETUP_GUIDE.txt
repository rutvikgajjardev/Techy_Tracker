નમસ્તે! આ ગાઇડ દ્વારા તમે તમારા પાયથોન (Flask) બેકએન્ડમાં કોઈપણ સિંગલ એક્સપેન્સ 
(Expense) માટે એક અત્યાધુનિક, મોર્ડન ડિજિટલ ઇન્વોઇસ/રસીદ (A4 PDF Receipt) જનરેટ 
કરતી API જાતે સ્ટેપ-બાય-સ્ટેપ બનાવી શકશો. 

આ PDF માં નીચેની તમામ બાબતો ડાયનેમિકલી સામેલ હશે:
1. Header & Branding: App Logo/Name, Expense Title અને Category.
2. Core Expense Card: Total Amount (Rs.), Paid By, Created Time & Last Updated Time.
3. Split Distribution: "Split Distribution (X Members)" હેઠળ ફક્ત સામેલ સભ્યોના નામોનું લિસ્ટ.
4. Attached Bills & Receipts: અપલોડ કરેલી અસલ બિલ તસવીરો (Cloudinary images) ને ઓટોમેટીક ફેચ કરીને એમ્બેડ કરવી.
5. Payment & Dynamic UPI QR Code: પૈસા મેળવનાર પેયરનો UPI ID અને સ્કેન કરી શકાય તેવો જીવંત UPI QR Code.

STEP 1: REQUIRED PYTHON LIBRARIES INSTALLATION

આ PDF API બનાવવા માટે આપણે 4 મુખ્ય Python પેકેજોની જરૂર પડશે:
- fpdf2    : અલ્ટ્રા-ફાસ્ટ, મોર્ડન અને પ્યોર પાયથોન PDF જનરેટર (In-memory support સાથે).
- qrcode   : ડાયનેમિક UPI QR કોડ જનરેટ કરવા માટે.
- pillow   : ઇમેજ પ્રોસેસિંગ અને ક્યુઆર કોડ રેન્ડરિંગ માટે.
- requests : ક્લાઉડિનરી (Cloudinary) પર સંગ્રહેલી બિલની તસવીરોને PDF માં લાવવા માટે.

class ExpenseReceiptPDF(FPDF):
    """
    કસ્ટમ પ્રોફેશનલ પીડીએફ ક્લાસ જે હેડર અને ફૂટર સંભાળે છે.
    """
    def header(self):
        # 1. ટોપ પ્રીમિયમ કલર બાર (Indigo Accent)
        self.set_fill_color(79, 70, 229) # Tailwind indigo-600
        self.rect(0, 0, 210, 5, style="F")

        # 2. બ્રાન્ડિંગ હેડર (Techy Tracker)
        self.set_xy(15, 12)
        self.set_font("Helvetica", "B", 18)
        self.set_text_color(30, 41, 59) # Slate-800
        self.cell(100, 8, "Techy Tracker", ln=0, align="L")

        self.set_font("Helvetica", "B", 10)
        self.set_text_color(99, 102, 241) # Indigo-500
        self.cell(80, 8, "DIGITAL EXPENSE RECEIPT", ln=1, align="R")

        self.set_x(15)
        self.set_font("Helvetica", "", 9)
        self.set_text_color(100, 116, 139) # Slate-500
        self.cell(100, 5, "Official Split & Settlement Statement", ln=0, align="L")
        
        # આડી ડિવાઈડર લાઈન
        self.set_draw_color(226, 232, 240)
        self.set_line_width(0.4)

        self.ln(16)

    def footer(self):
        self.set_y(-18)
        self.set_draw_color(226, 232, 240)
        self.set_line_width(0.3)
        self.line(15, self.get_y(), 195, self.get_y())
        self.set_y(-14)
        self.set_font("Helvetica", "", 8)
        self.set_text_color(148, 163, 184)
        self.cell(90, 8, "Generated securely via Techy Tracker Web App", align="L")
        self.cell(90, 8, f"Page {self.page_no()}", align="R")


@expense_bp.route("/<int:expense_id>/pdf", methods=["GET"])
def download_expense_pdf(expense_id):
    """
    ડાયનેમિક સિંગલ એક્સપેન્સ પીડીએફ રસીદ જનરેટર:
    - ડેટાબેઝમાંથી સંપૂર્ણ એક્સપેન્સ, કેટેગરી, પેયર, પાર્ટિસિપન્ટ્સ અને બિલ્સ ફેચ કરે છે.
    - મેમરીમાં (In-Memory) QR કોડ અને પીડીએફ બનાવીને ક્લાયન્ટને સેન્ડ કરે છે.
    """
    import asyncio

    async def _generate():
        db = Prisma()
        await db.connect()
        try:
            # પ્રિઝમામાંથી જરૂરી તમામ રિલેશન્સ સાથે એક્સપેન્સ ફેચ કરો
            exp = await db.expense.find_unique(
                where={"id": expense_id},
                include={
                    "category": True,
                    "paidBy": True,
                    "createdBy": True,
                    "participants": {"include": {"user": True}},
                }
            )
            if not exp:
                return {"error": "Expense not found", "status": 404}

            payer_name = exp.paidBy.name if exp.paidBy else "Unknown"
            payer_upi = getattr(exp.paidBy, "upi_id", None) or ""
            cat_name = exp.category.name if exp.category else "General"
            total_amount = float(exp.amount)
            
            # ટાઇમસ્ટેમ્પ ફોર્મેટિંગ
            created_str = exp.createdAt.strftime("%d %b %Y, %I:%M %p")
            updated_str = exp.updatedAt.strftime("%d %b %Y, %I:%M %p")

            # -----------------------------------------------------------------
            # 1. PDF ડોક્યુમેન્ટ તૈયાર કરવું (A4 Portrait)
            # -----------------------------------------------------------------
            pdf = ExpenseReceiptPDF(orientation="P", unit="mm", format="A4")
            pdf.set_auto_page_break(auto=True, margin=20)
            pdf.add_page()

            # -----------------------------------------------------------------
            # 2. ટાઇટલ અને કેટેગરી બેજ (Header Metadata)
            # -----------------------------------------------------------------
            pdf.set_fill_color(241, 245, 249) # Slate-100 badge
            pdf.set_text_color(79, 70, 229)   # Indigo text
            pdf.set_font("Helvetica", "B", 9)
            pdf.cell(32, 6, f" {cat_name.upper()} ", ln=1, align="C", fill=True)
            pdf.ln(2)

            pdf.set_font("Helvetica", "B", 16)
            pdf.set_text_color(15, 23, 42)    # Slate-900
            pdf.cell(0, 8, exp.title, ln=1)

            if exp.description and exp.description.strip():
                pdf.set_font("Helvetica", "", 9)
                pdf.set_text_color(100, 116, 139)
                pdf.multi_cell(180, 5, exp.description.strip())
            pdf.ln(4)

            # -----------------------------------------------------------------
            # 3. મુખ્ય કાર્ડ (Core Expense Summary Card)
            # -----------------------------------------------------------------
            card_x = 15
            card_y = pdf.get_y()
            card_w = 180
            card_h = 32

            # કાર્ડ બેકગ્રાઉન્ડ અને બોર્ડર
            pdf.set_fill_color(248, 250, 252) # Slate-50
            pdf.set_draw_color(203, 213, 225) # Slate-300
            pdf.rect(card_x, card_y, card_w, card_h, style="DF")

            # ડાબી બાજુ: કુલ રકમ (Total Amount)
            pdf.set_xy(card_x + 6, card_y + 4)
            pdf.set_font("Helvetica", "B", 8)
            pdf.set_text_color(100, 116, 139)
            pdf.cell(80, 4, "TOTAL AMOUNT", ln=1)

            pdf.set_xy(card_x + 6, card_y + 10)
            pdf.set_font("Helvetica", "B", 20)
            pdf.set_text_color(15, 23, 42)
            # સ્ટાન્ડર્ડ હેલ્વેટિકા ફોન્ટ માટે "Rs." વાપરો જેથી એન્કોડિંગ એરર ન આવે
            pdf.cell(80, 12, f"Rs. {total_amount:,.2f}", ln=1)

            # જમણી બાજુ: પેયર અને ટાઇમસ્ટેમ્પ્સ
            info_x = card_x + 95
            pdf.set_xy(info_x, card_y + 4)
            pdf.set_font("Helvetica", "B", 8)
            pdf.set_text_color(100, 116, 139)
            pdf.cell(30, 4, "Expense By (Paid):", ln=0)
            pdf.set_font("Helvetica", "B", 9)
            pdf.set_text_color(15, 23, 42)
            pdf.cell(50, 4, payer_name, ln=1)

            pdf.set_xy(info_x, card_y + 12)
            pdf.set_font("Helvetica", "", 8)
            pdf.set_text_color(100, 116, 139)
            pdf.cell(30, 4, "Created Time:", ln=0)
            pdf.set_text_color(51, 65, 85)
            pdf.cell(50, 4, created_str, ln=1)

            pdf.set_xy(info_x, card_y + 19)
            pdf.set_font("Helvetica", "", 8)
            pdf.set_text_color(100, 116, 139)
            pdf.cell(30, 4, "Last Updated:", ln=0)
            pdf.set_text_color(51, 65, 85)
            pdf.cell(50, 4, updated_str, ln=1)

            pdf.set_y(card_y + card_h + 8)

            # -----------------------------------------------------------------
            # 4. Split Distribution (Strictly ONLY List of Member Names)
            # -----------------------------------------------------------------
            member_names = [p.user.name for p in exp.participants if p.user]
            member_count = len(member_names)

            pdf.set_font("Helvetica", "B", 11)
            pdf.set_text_color(15, 23, 42)
            pdf.cell(0, 6, f"Split Distribution ({member_count} Members)", ln=1)

            pdf.set_draw_color(226, 232, 240)
            pdf.set_line_width(0.3)
            pdf.line(15, pdf.get_y(), 195, pdf.get_y())
            pdf.ln(3)

            # દરેક સભ્યનું નામ સ્વચ્છ બુલેટ પોઇન્ટ લિસ્ટમાં દર્શાવો
            pdf.set_font("Helvetica", "", 9)
            pdf.set_text_color(51, 65, 85)
            
            # સભ્યોના નામોને 2-કોલમ ગ્રીડમાં ગોઠવીએ છીએ જેથી જગ્યા બચે
            col_w = 90
            for idx, name in enumerate(member_names):
                is_right_col = (idx % 2 == 1)
                if not is_right_col and idx > 0:
                    pdf.ln(5)
                
                # પેયરની બાજુમાં (Payer) ટેગ દર્શાવો
                tag = " (Payer)" if name == payer_name else ""
                pdf.cell(col_w, 5, f"-  {name}{tag}", ln=1 if is_right_col else 0)

            pdf.ln(8)

            # -----------------------------------------------------------------
            # 5. Attached Bills & Receipts (બિલની તસવીરો ફેચ અને એમ્બેડ કરવી)
            # -----------------------------------------------------------------
            raw_images = getattr(exp, "images", []) or []
            valid_images = [img for img in raw_images if img and str(img).startswith("http")]

            if valid_images:
                pdf.set_font("Helvetica", "B", 11)
                pdf.set_text_color(15, 23, 42)
                pdf.cell(0, 6, f"Attached Bills & Receipts ({len(valid_images)})", ln=1)
                pdf.line(15, pdf.get_y(), 195, pdf.get_y())
                pdf.ln(4)

                img_w = 55  # mm પહોળાઈ
                for img_url in valid_images[:3]: # મહત્તમ 3 તસવીરો પ્રદર્શિત કરો
                    try:
                        resp = requests.get(img_url, timeout=5)
                        if resp.status_code == 200:
                            img_buf = io.BytesIO(resp.content)
                            with Image.open(img_buf) as p_img:
                                w, h = p_img.size
                                img_h = min(round((h / max(w, 1)) * img_w, 2), 65)

                            # જો પાનાના છેડે જગ્યા ઓછી હોય તો નવું પેજ ઉમેરો
                            if pdf.get_y() + img_h > 260:
                                pdf.add_page()

                            cur_y = pdf.get_y()
                            # ઇમેજ ફ્રેમ અને શેડો
                            pdf.set_draw_color(226, 232, 240)
                            pdf.rect(15, cur_y, img_w, img_h)
                            img_buf.seek(0)
                            pdf.image(img_buf, x=15.5, y=cur_y + 0.5, w=img_w - 1, h=img_h - 1)
                            pdf.set_y(cur_y + img_h + 4)
                    except Exception as img_err:
                        # જો ઇન્ટરનેટ કે લિંકમાં વાંધો હોય તો ફેલ ન થવું જોઈએ
                        pass
                pdf.ln(4)

            # -----------------------------------------------------------------
            # 6. Payment Section (Scan to Pay & Dynamic UPI QR Code)
            # -----------------------------------------------------------------
            # જો છેલ્લે QR કોડ માટે જગ્યા ઓછી હોય તો નવું પેજ
            if pdf.get_y() > 220:
                pdf.add_page()

            pay_box_x = 15
            pay_box_y = pdf.get_y()
            pay_box_w = 180
            pay_box_h = 36

            pdf.set_fill_color(248, 250, 252) # Slate-50
            pdf.set_draw_color(203, 213, 225)
            pdf.rect(pay_box_x, pay_box_y, pay_box_w, pay_box_h, style="DF")

            if payer_upi:

                encoded_payer = urllib.parse.quote(payer_name)

                qr_buf.seek(0)

                pdf.image(qr_buf, x=pay_box_x + 5, y=pay_box_y + 3, w=30, h=30)

                pdf.set_xy(pay_box_x + 40, pay_box_y + 5)

                pdf.set_text_color(100, 116, 139)
                pdf.cell(160, 6, f"Note: {payer_name} has not registered a UPI ID yet. Settle manually via Cash/Bank.", align="C")

            # -----------------------------------------------------------------
            # 7. Output PDF bytes & Send File
            # -----------------------------------------------------------------
            pdf_bytes = bytes(pdf.output())
            return {
                "success": True,
